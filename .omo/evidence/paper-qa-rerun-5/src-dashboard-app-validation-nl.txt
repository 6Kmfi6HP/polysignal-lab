   220	        if value is None or value == "":
   221	            continue
   222	        try:
   223	            return float(str(value))
   224	        except (TypeError, ValueError):
   225	            continue
   226	    return None

   229	def _market_text(market: Market | None, key: str) -> str:
   230	    if market is None:
   231	        return ""
   232	    value = getattr(market, key, "")
   233	    return "" if value is None else str(value)

   236	def _resolve_side(row: dict[str, JsonValue], market: Market | None, *, token_id: str) -> str:
   237	    metrics = _metrics_of(row)
   238	    for candidate in (
   239	        row.get("side"),
   240	        row.get("order_side"),
   241	        metrics.get("side"),
   242	    ):
   243	        upper = "" if candidate in (None, "") else str(candidate).upper()
   244	        if upper == "UP":
   245	            return "UP"
   246	        if upper == "DOWN":
   247	            return "DOWN"
   248	    if market is not None:
   249	        for token in market.outcome_tokens:
   250	            if token.token_id == token_id:
   251	                return str(token.side)
   252	    return ""

   255	def _normalize_position_status(row: dict[str, JsonValue]) -> str:
   256	    raw = _row_text(row, "status", metric_keys=("status",))
   257	    upper = raw.upper()
   258	    if upper == "OPEN":
   259	        return "OPEN"
   260	    if upper == "CLOSED":
   261	        return "CLOSED"
   262	    if bool(row.get("is_closed")):
   263	        return "CLOSED"
   264	    return ""

   267	def _normalize_order_status(raw: str) -> str:
   268	    upper = raw.upper()
   269	    for group, mapped in (
   270	        ({"FILLED"}, "FILLED"),
   271	        ({"PARTIAL", "PARTIALLY_FILLED"}, "PARTIAL"),
   272	        ({"REJECTED", "DENIED"}, "REJECTED"),
   273	        ({"CANCELLED", "CANCELED", "EXPIRED"}, "CANCELLED"),
   274	        ({"ACCEPTED", "RESTING", "SUBMITTED"}, "RESTING"),
   275	    ):
   276	        if upper in group:
   277	            return mapped
   278	    return ""

   281	def _finite_nonnegative(value: JsonValue) -> bool:
   282	    if value in (None, ""):
   283	        return False
   284	    try:
   285	        parsed = float(str(value))
   286	    except (TypeError, ValueError):
   287	        return False
   288	    return math.isfinite(parsed) and parsed >= 0.0

   291	def _valid_timestamp(value: JsonValue) -> bool:
   292	    if isinstance(value, datetime):
   293	        return True
   294	    if not isinstance(value, str):
   295	        return False
   296	    try:
   297	        _ = datetime.fromisoformat(value.replace("Z", "+00:00"))
   298	    except ValueError:
   299	        return False
   300	    return True

   303	def _valid_order_payload(payload: dict[str, JsonValue]) -> bool:
   304	    return bool(payload.get("paper_order_id")) and bool(payload.get("status"))

   307	def _valid_position_payload(payload: dict[str, JsonValue]) -> bool:
   308	    if not bool(payload.get("paper_position_id")):
   309	        return False
   310	    if payload.get("side") not in {"UP", "DOWN"}:
   311	        return False
   312	    if payload.get("status") not in {"OPEN", "CLOSED"}:
   313	        return False
   314	    opened_at = payload.get("opened_at")
   315	    if opened_at in (None, ""):
   316	        return False
   317	    if not _valid_timestamp(opened_at):
   318	        return False
   319	    for key in ("entry_price", "shares", "stake_usdc"):
   320	        value = payload.get(key)
   321	        if value in (None, ""):
   322	            return False
   323	        if not _finite_nonnegative(value):
   324	            return False
   325	    return True

   362	def _paper_position_payload(
   363	    row: dict[str, JsonValue],
   364	    *,
   365	    market: Market | None,
   366	) -> dict[str, JsonValue]:
   367	    payload = dict(row)
   368	    position_id = _row_text(row, "position_id", "paper_position_id")
   369	    _fill_missing(payload, "paper_position_id", position_id)
   370	    _fill_missing(payload, "position_id", position_id)
   371	    _fill_missing(payload, "signal_id", _row_text(row, "signal_id", metric_keys=("signal_id",)))
   372	    _fill_missing(payload, "paper_order_id", _row_text(row, "paper_order_id", "client_order_id", metric_keys=("paper_order_id", "client_order_id")))
   373	    _fill_missing(payload, "paper_fill_id", _row_text(row, "paper_fill_id", "trade_id", metric_keys=("paper_fill_id", "trade_id")))
   374	    _fill_missing(payload, "strategy", _row_text(row, "strategy", metric_keys=("strategy",)))
   375	    _fill_missing(payload, "asset", _row_text(row, "asset", metric_keys=("asset",)) or _market_text(market, "asset"))
   376	    _fill_missing(payload, "timeframe", _row_text(row, "timeframe", metric_keys=("timeframe",)) or _market_text(market, "timeframe"))
   377	    _fill_missing(payload, "market_id", _row_text(row, "market_id", metric_keys=("market_id",)) or _market_text(market, "market_id"))
   378	    _fill_missing(payload, "market_slug", _row_text(row, "market_slug", metric_keys=("market_slug",)) or _market_text(market, "market_slug"))
   379	    token_id = _instrument_token_id(_row_text(row, "instrument_id", "token_id", metric_keys=("token_id",))) or _row_text(row, "token_id")
   380	    payload["token_id"] = token_id
   381	    payload["side"] = _resolve_side(row, market, token_id=token_id)
   382	    entry_price = _row_number(row, "entry_price", "avg_entry_price", "price", "last_px", metric_keys=("entry_price", "avg_entry_price", "price"))
   383	    _fill_missing(payload, "entry_price", entry_price)
   384	    shares = _row_number(row, "shares", "quantity", "signed_qty", metric_keys=("shares", "quantity"))
   385	    _fill_missing(payload, "shares", shares)
   386	    stake = _row_number(row, "stake_usdc", metric_keys=("stake_usdc",))
   387	    if stake is None and entry_price is not None and shares is not None:
   388	        stake = entry_price * abs(shares)
   389	    _fill_missing(payload, "stake_usdc", stake)
   390	    _fill_missing(payload, "opened_at", _row_text(row, "opened_at", "ts", "created_at"))
   391	    payload["status"] = _normalize_position_status(row)
   392	    _fill_missing(payload, "closed_at", _row_text(row, "closed_at", metric_keys=("closed_at",)))
   393	    payload["is_closed"] = payload["status"] == "CLOSED"
   394	    return payload
