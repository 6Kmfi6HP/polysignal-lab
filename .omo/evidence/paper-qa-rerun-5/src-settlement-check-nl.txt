   157	async def _try_settle_projection(
   158	    scheduler: _SettlementScheduler, projection: dict[str, object]
   159	) -> dict[str, Any] | None:
   160	    """Try to settle a single position projection.
   161	
   162	    Returns the settlement result, or None if the projection should
   163	    not be settled (e.g., already closed, no market found, unready).
   164	    """
   165	    if not isinstance(projection, dict):
   166	        return None
   167	    if bool(projection.get("is_closed")):
   168	        return None
   169	    market_id = str(projection.get("market_id") or "")
   170	    token_id = str(projection.get("token_id") or projection.get("instrument_id") or "")
   171	    if not market_id or not token_id:
   172	        return None
   173	    market = _load_market(scheduler, market_id)
   174	    if market is None:
   175	        return None
   176	    settlement_resolver = getattr(scheduler, "settlement_resolver", None)
   177	    if settlement_resolver is None:
   178	        return None
   179	    decision = await settlement_resolver.resolve_market(market)
   180	    outcome_value: float | None
   181	    if decision.status == "resolved":
   182	        outcome_value = decision.outcome_value_for(token_id)
   183	    elif decision.status == "cancelled":
   184	        outcome_value = _projection_float(projection, "avg_entry_price")
   185	        if outcome_value is None:
   186	            outcome_value = _projection_float(projection, "entry_price")
   187	    else:
   188	        return None
   189	    if outcome_value is None:
   190	        return None
   191	    paper_position_id = str(
   192	        projection.get("paper_position_id") or projection.get("position_id") or ""
   193	    )
   194	    if not paper_position_id:
   195	        return None
   196	    if _existing_result_for_position(scheduler, paper_position_id):
   197	        return None
   198	    result = _paper_trade_result_from_projection(
   199	        projection,
   200	        market=market,
   201	        outcome_value=outcome_value,
   202	        details=dict(decision.details),
   203	    )
   204	    if result is None:
   205	        return None
   206	    await _store_projection_result(scheduler, result)
   207	    if decision.conflict:
   208	        await _log_settlement_conflict(scheduler, decision, result)
   209	    return result
