    62	def _valid_position_event(row: Mapping[str, Any]) -> bool:
    63	    position_id = str(row.get("paper_position_id") or row.get("position_id") or "")
    64	    if not position_id:
    65	        return False
    66	    status = str(row.get("status") or "").upper()
    67	    if status not in {"", PositionStatus.OPEN.value, PositionStatus.CLOSED.value}:
    68	        return False
    69	    if status == "" and not isinstance(row.get("is_closed"), bool):
    70	        return False
    71	    is_open = status == PositionStatus.OPEN.value or (
    72	        status == "" and row.get("is_closed") is False
    73	    )
    74	    side = str(row.get("side") or "").upper()
    75	    if is_open and side not in {Side.UP.value, Side.DOWN.value}:
    76	        return False
    77	    if is_open and (
    78	        _row_finite_float(row, "shares", "quantity", "signed_qty") is None
    79	        or _row_finite_float(row, "entry_price", "avg_entry_price") is None
    80	        or _row_finite_float(row, "stake_usdc") is None
    81	        or _row_timestamp(row, "opened_at", "ts", "created_at") is None
    82	    ):
    83	        return False
    84	    for key in ("shares", "quantity", "signed_qty", "entry_price", "avg_entry_price", "stake_usdc"):
    85	        value = row.get(key)
    86	        if value in (None, ""):
    87	            continue
    88	        parsed = _row_finite_float(row, key)
    89	        if parsed is None or parsed < 0.0:
    90	            return False
    91	    return True

   100	            return None
