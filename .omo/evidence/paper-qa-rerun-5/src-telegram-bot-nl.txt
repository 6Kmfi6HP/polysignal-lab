   113	    async def start(self) -> None:
   114	        if not self.config.interactive_enabled:
   115	            return
   116	        if self.config.interactive_dry_run:
   117	            self.logger.info("Telegram interactive bot interactive_dry_run enabled")
   118	        if not self.config.resolved_bot_token:
   119	            self._error = "missing bot token"
   120	            self.logger.warning("Telegram interactive bot disabled: missing bot token")
   121	            return
   122	        if not self.config.interactive_allowed_chat_ids:
   123	            self._error = "no allowed chat ids"
   124	            self.logger.warning("Telegram interactive bot disabled: no allowed chat ids")
   125	            return
   126	        self.application = self.application or (
   127	            ApplicationBuilder()
   128	            .token(self.config.resolved_bot_token)
   129	            .rate_limiter(AIORateLimiter(max_retries=self.config.retry_attempts))
   130	            .concurrent_updates(True)
   131	            .build()
   132	        )
   133	        self.configure_handlers()
   134	        try:
   135	            await self.application.initialize()
   136	            await self._set_bot_commands()
   137	            await self.application.start()
   138	            if self.application.updater is None:
   139	                raise RuntimeError("telegram application updater is not available")
   140	            await self.application.updater.start_polling(
   141	                poll_interval=self.config.interactive_poll_interval_sec,
   142	                timeout=self.config.interactive_poll_timeout_sec,
   143	                allowed_updates=("message", "callback_query"),
   144	                drop_pending_updates=self.config.interactive_drop_pending_updates_on_start,
   145	            )
   146	        except RetryAfter as exc:
   147	            self._rate_limited += 1
   148	            self._poll_failure += 1
   149	            self._error = f"rate limited: retry after {exc.retry_after}s"
   150	            self.logger.warning("Telegram interactive bot rate limited")
   151	            return
   152	        except (TimedOut, NetworkError, TelegramError, RuntimeError) as exc:
   153	            self._poll_failure += 1
   154	            self._error = type(exc).__name__
   155	            self.logger.warning("Telegram interactive bot start failed: %s", type(exc).__name__)
   156	            return
   157	        self._running = True
   158	        self._poll_success += 1
   159	        self._error = None

   172	    def _authorized(self, update: Update) -> bool:
   173	        chat = update.effective_chat
   174	        user = update.effective_user
   175	        allowed_ids = set(self.config.interactive_allowed_chat_ids)
   176	        allowed = (
   177	            chat is not None
   178	            and user is not None
   179	            and str(chat.type) == ChatType.PRIVATE
   180	            and int(chat.id) in allowed_ids
   181	            and int(user.id) in allowed_ids
   182	        )
   183	        if allowed:
   184	            self._last_update_id = getattr(update, "update_id", None)
   185	            self._last_update_at = utc_iso()
   186	            return True
   187	        self._unauthorized_updates += 1
   188	        self.logger.warning(
   189	            "Unauthorized Telegram interactive update chat_id=%s user_id=%s",
   190	            getattr(chat, "id", None),
   191	            getattr(user, "id", None),
   192	        )
   193	        return False

   281	    async def _callback(self, update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
   282	        query = update.callback_query
   283	        if query is None:
   284	            return
   285	        if not self._authorized(update):
   286	            await query.answer("Unauthorized", show_alert=True)
   287	            return
   288	        data = query.data or ""
   289	        known_actions = {"m", "bk", "p", "st", "sg", "dy", "str", "lb", "lbt"}
   290	        if data.startswith("tg:"):
   291	            name = data[3:]
   292	            if name not in set(self._strategy_names()):
   293	                await query.answer("Unknown strategy", show_alert=True)
   294	                return
   295	        elif data not in known_actions:
   296	            await query.answer("Unknown action", show_alert=True)
   297	            return
   298	        try:
   299	            await query.answer()
   300	        except (TimedOut, NetworkError, TelegramError):
   301	            self._send_failure += 1
   302	        try:
   303	            if data.startswith("tg:"):
   304	                text, keyboard = self._toggle_strategy(data)
   305	            else:
   306	                text, keyboard = self._render_callback(data)
   307	        except Exception:
   308	            await self._edit_or_reply(update, "Action failed", self._back_keyboard())
   309	            self.logger.exception("Telegram callback failed")
   310	            return
   311	        await self._edit_or_reply(update, text, keyboard)

   336	    def _format_positions(self) -> str:
   337	        rows = self.persistence.restore_open_positions()
   338	        if not rows:
   339	            return "暂无 open paper positions。"
   340	        blocks: list[str] = []
   341	        for row in rows:
   342	            payload = _position_display_payload(row)
   343	            if not payload.get("side"):
   344	                continue
   345	            token_id = str(payload.get("token_id") or "")
   346	            book = self.books.get(token_id) if token_id else None
   347	            mark = book.best_bid if book is not None else None
   348	            entry_price = float(payload.get("entry_price") or 0.0)
   349	            shares = float(payload.get("shares") or 0.0)
   350	            stake_usdc = float(payload.get("stake_usdc") or 0.0)
   351	            lines = [
   352	                f"📈 {self._safe(payload.get('asset'))} {self._safe(payload.get('timeframe'))} · {self._safe(payload.get('side'))}",
   353	                f"Strategy  {self._safe(payload.get('strategy'))}",
   354	                f"Entry     {entry_price:.4f}",
   355	            ]
   356	            if mark is None:
   357	                lines.extend(["Mark      n/a (live book unavailable)", "PnL       n/a"])
   358	            else:
   359	                pnl = (mark - entry_price) * shares
   360	                roi = pnl / stake_usdc if stake_usdc else 0.0
   361	                sign = "+" if pnl >= 0 else ""
   362	                lines.extend(
   363	                    [
   364	                        f"Mark      {mark:.4f}",
   365	                        f"Shares    {shares:.4f}",
   366	                        f"PnL       {sign}{pnl:.2f} USDC ({sign}{roi:.2%})",
   367	                    ]
   368	                )
   369	            lines.extend(
   370	                [
   371	                    f"Opened    {self._format_age(payload.get('opened_at'))}",
   372	                    f"ID        {self._safe(payload.get('paper_position_id'))}",
   373	                ]
   374	            )
   375	            blocks.append("\n".join(lines))
   376	        if not blocks:
   377	            return "暂无 open paper positions。"
   378	        return "\n\n".join(blocks)

   380	    def _format_signals(self) -> str:
   381	        accepted = self.persistence.query_json(
   382	            "signals", where="ORDER BY created_at DESC", limit=5
   383	        )
   384	        rejected = self.persistence.query_json(
   385	            "rejected_signals", where="ORDER BY rejected_at DESC", limit=5
   386	        )
   387	        items: list[tuple[datetime, str]] = []
   388	        for row in accepted:
   389	            ts = self._parse_time(row.get("created_at")) or datetime.min.replace(tzinfo=timezone.utc)
   390	            items.append((ts, self._format_accepted_signal(row)))
   391	        for row in rejected:
   392	            ts = self._parse_time(row.get("rejected_at")) or datetime.min.replace(tzinfo=timezone.utc)
   393	            items.append((ts, self._format_rejected_signal(row)))
   394	        if not items:
   395	            return "暂无 recent signals。"
   396	        return "\n\n".join(text for _, text in sorted(items, key=lambda item: item[0], reverse=True)[:5])

   446	    def _today_closed_bounds(self) -> tuple[datetime, datetime]:
   447	        report_tz = self._report_timezone()
   448	        today = datetime.now(report_tz).date()
   449	        day_start_local = datetime.combine(today, time.min, tzinfo=report_tz)
