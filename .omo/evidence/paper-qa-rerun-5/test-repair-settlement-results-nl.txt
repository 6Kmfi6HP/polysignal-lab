    36	def test_settle_for_repair_returns_parseable_trade_result_row() -> None:
    37	    market = sample_market()
    38	    token = market.token_for(Side.UP)
    39	    position = {
    40	        "paper_position_id": "pos-repair-1",
    41	        "position_id": "pos-repair-1",
    42	        "signal_id": "sig-repair-1",
    43	        "strategy": "ptb_diff",
    44	        "asset": "BTC",
    45	        "timeframe": "5m",
    46	        "market_id": market.market_id,
    47	        "market_slug": market.market_slug,
    48	        "token_id": token.token_id,
    49	        "side": Side.UP.value,
    50	        "entry_price": 0.4,
    51	        "shares": 25.0,
    52	        "stake_usdc": 10.0,
    53	        "opened_at": "2026-06-22T00:00:00+00:00",
    54	    }
    55	    decision = ResolutionDecision(
    56	        market_id=market.market_id,
    57	        condition_id=market.condition_id,
    58	        status="resolved",
    59	        source="chain",
    60	        outcome_values_by_token={token.token_id: 1.0},
    61	        conflict=False,
    62	        conflict_sources=(),
    63	        details={},
    64	    )
    65	
    66	    module = _repair_module()
    67	    result = module._settle_for_repair(position, market, decision)
    68	
    69	    assert result is not None
    70	    parsed = parse_paper_trade_result_row(result)
    71	    assert parsed.get("paper_trade_id")
    72	    assert parsed.get("signal_id") == "sig-repair-1"
