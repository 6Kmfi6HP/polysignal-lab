   125	    rows = runtime.sqlite.query_json(

   137	    import polysignal_lab.app._settlement_check as settlement_mod
   138	
   139	    runtime = _runtime(tmp_path)

   141	        MarketFactoryConfig(asset="BTC", timeframe="5m", seconds_to_close=-1)

   145	        ResolutionDecision(
   146	            cancelled_market.market_id,

   157	    scheduler.sqlite = runtime.sqlite

   159	    runtime.markets.upsert_many([cancelled_market])
   160	
   161	    monkeypatch.setattr(
   162	        settlement_mod,
   163	        "_nautilus_positions",
   164	        lambda s: [
   165	            _projection(
   166	                cancelled_market.market_id,

   168	            )
   169	        ],
   170	    )
   171	
   172	    results = await check_settlements(scheduler)
   173	
   174	    result_rows = runtime.sqlite.query_json("paper_trade_results")
   175	    assert [result["result"] for result in results] == [TradeResultStatus.VOID.value]
