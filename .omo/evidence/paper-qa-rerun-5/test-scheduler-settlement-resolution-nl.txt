   192	@pytest.mark.anyio
   193	async def test_check_settlements_is_idempotent_per_position(monkeypatch) -> None:
   194	    _patch_positions(monkeypatch)
   195	    stored: list[dict[str, object]] = []
   196	
   197	    def query_json(table: str, **kwargs: object) -> list[dict[str, object]]:
   198	        if table == "paper_trade_results":
   199	            return list(stored)
   200	        return []
   201	
   202	    def insert_paper_trade_result(result: Mapping[str, object]) -> None:
   203	        stored.append(dict(result))
   204	
   205	    scheduler = _scheduler(
   206	        _market(),
   207	        ResolutionDecision(
   208	            "market-1",
   209	            "0x" + "1" * 64,
   210	            "resolved",
   211	            "chain",
   212	            {"token-up": 1.0, "token-down": 0.0},
   213	            False,
   214	            (),
   215	            {"settlement_source": "chain"},
   216	        ),
   217	    )
   218	    scheduler.persistence.query_json = query_json
   219	    scheduler.persistence.insert_paper_trade_result = insert_paper_trade_result
   220	
   221	    first = await check_settlements(scheduler)
   222	    second = await check_settlements(scheduler)
   223	
   224	    assert len(first) == 1
   225	    assert second == []
   226	    assert len(stored) == 1
   227	    assert stored[0]["paper_position_id"] == "pos-1"
